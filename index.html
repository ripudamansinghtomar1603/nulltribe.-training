const express = require('express'); const cors = require('cors'); const jwt = require('jsonwebtoken'); const PDFDocument = require('pdfkit'); const app = express(); // Enable CORS for frontend communication app.use(cors()); app.use(express.json()); const SECRET_KEY = 'your-super-secret-key-change-this-in-production'; // --- MOCK DATABASE --- const users = []; const progressDb = {}; // --- AUTHENTICATION MIDDLEWARE --- const authenticateToken = (req, res, next) => { const authHeader = req.headers['authorization']; const token = authHeader && authHeader.split(' ')[1]; if (!token) { return res.status(401).json({ error: 'Access denied. Authentication token missing.' }); } jwt.verify(token, SECRET_KEY, (err, user) => { if (err) { return res.status(403).json({ error: 'Invalid or expired session token. Please log in again.' }); } req.user = user; next(); }); }; // --- API ROUTES --- app.post('/api/signup', (req, res) => { const { name, email, password } = req.body; if (!name || !email || !password) { return res.status(400).json({ error: 'All fields are required.' }); } if (users.find(u => u.email.toLowerCase() === email.toLowerCase())) { return res.status(400).json({ error: 'Account already exists.' }); } const newUser = { id: Date.now().toString(), name: name.trim(), email: email.toLowerCase().trim(), password }; users.push(newUser); progressDb[newUser.id] = [false, false, false, false]; const token = jwt.sign({ id: newUser.id, name: newUser.name }, SECRET_KEY, { expiresIn: '24h' }); res.status(201).json({ token, user: { name: newUser.name, email: newUser.email }, progress: progressDb[newUser.id] }); }); app.post('/api/login', (req, res) => { const { email, password } = req.body; if (!email || !password) { return res.status(400).json({ error: 'Email and password are required.' }); } const user = users.find(u => u.email.toLowerCase() === email.toLowerCase().trim() && u.password === password); if (!user) { return res.status(400).json({ error: 'Invalid email or password.' }); } const token = jwt.sign({ id: user.id, name: user.name }, SECRET_KEY, { expiresIn: '24h' }); res.json({ token, user: { name: user.name, email: user.email }, progress: progressDb[user.id] || [false, false, false, false] }); }); app.post('/api/progress', authenticateToken, (req, res) => { const { scenarioIndex } = req.body; if (scenarioIndex === undefined || scenarioIndex < 0 || scenarioIndex > 3) { return res.status(400).json({ error: 'Invalid scenario index.' }); } if (!progressDb[req.user.id]) { progressDb[req.user.id] = [false, false, false, false]; } progressDb[req.user.id][scenarioIndex] = true; res.json({ success: true, progress: progressDb[req.user.id] }); }); /** * @route GET /api/certificate * @desc Generate fixed, absolutely positioned PDF certificate */ app.get('/api/certificate', authenticateToken, (req, res) => { const userProgress = progressDb[req.user.id]; if (!userProgress || !userProgress.every(Boolean)) { return res.status(403).json({ error: 'Complete all four training scenarios to unlock your certificate.' }); } // Set margins to 0 to prevent PDFKit from triggering automatic blank pages const doc = new PDFDocument({ layout: 'landscape', size: 'A4', margins: { top: 0, bottom: 0, left: 0, right: 0 } }); const sanitizedFilename = req.user.name.replace(/[^a-z0-9]/gi, '_').toLowerCase(); res.setHeader('Content-Type', 'application/pdf'); res.setHeader('Content-Disposition', `attachment; filename="nulltribe_${sanitizedFilename}_certificate.pdf"`); doc.pipe(res); const pageWidth = doc.page.width; const pageHeight = doc.page.height; // --- DRAW CERTIFICATE --- // Geometric Borders doc.rect(25, 20, pageWidth - 50, pageHeight - 40).lineWidth(3).stroke('#3B65E8'); doc.rect(32, 27, pageWidth - 64, pageHeight - 54).lineWidth(1).stroke('#E4E8F0'); // Exact Width Calculation for perfectly centered 3-color Logo doc.fontSize(28).font('Helvetica-Bold'); const word1 = doc.widthOfString('null'); const word2 = doc.widthOfString('tribe'); const word3 = doc.widthOfString('.'); const logoStartX = (pageWidth - (word1 + word2 + word3)) / 2; doc.fillColor('#3B65E8').text('null', logoStartX, 85, { lineBreak: false }); doc.fillColor('#1A1F2E').text('tribe', logoStartX + word1, 85, { lineBreak: false }); doc.fillColor('#F97316').text('.', logoStartX + word1 + word2, 85, { lineBreak: false }); // Sub-eyebrow doc.fontSize(10).font('Helvetica-Bold').fillColor('#8892A4') .text('SECURITY ORCHESTRATOR PLATFORM', 0, 125, { align: 'center', width: pageWidth, letterSpacing: 2 }); // Certificate Title doc.fontSize(38).font('Helvetica-Bold').fillColor('#1A1F2E') .text('Certificate of Completion', 0, 185, { align: 'center', width: pageWidth }); doc.fontSize(15).font('Helvetica').fillColor('#4A5568') .text('This credential validates that', 0, 255, { align: 'center', width: pageWidth }); // Recipient Name doc.fontSize(32).font('Helvetica-Bold').fillColor('#3B65E8') .text(req.user.name, 0, 295, { align: 'center', width: pageWidth }); // Manual Underline const nameWidth = doc.widthOfString(req.user.name); doc.moveTo((pageWidth / 2) - (nameWidth / 2), 335) .lineTo((pageWidth / 2) + (nameWidth / 2), 335) .lineWidth(1.5).stroke('#3B65E8'); // Course Details doc.fontSize(14).font('Helvetica').fillColor('#4A5568') .text('has successfully reviewed and mastered all interactive modules for:', 0, 370, { align: 'center', width: pageWidth }); doc.fontSize(20).font('Helvetica-Bold').fillColor('#1A1F2E') .text('Information Security Risk Fundamentals — Phase 1', 0, 400, { align: 'center', width: pageWidth }); // Footer Metadata Blocks const issueDate = new Date().toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' }); const bottomY = 485; // Left: Date Block doc.fontSize(12).font('Helvetica-Bold').fillColor('#1A1F2E').text(`Date of Issuance:`, 80, bottomY); doc.fontSize(12).font('Helvetica').fillColor('#4A5568').text(issueDate, 80, bottomY + 18); // Right: Issuer Authority Block doc.fontSize(12).font('Helvetica-Bold').fillColor('#1A1F2E') .text('Issued & Verified by:', pageWidth - 320, bottomY, { align: 'right', width: 240 }); doc.fontSize(11).font('Helvetica').fillColor('#8892A4') .text('nulltribe. Application Security Portal', pageWidth - 320, bottomY + 18, { align: 'right', width: 240 }); doc.end(); }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => console.log(`[nulltribe API] Orchestrator online at port ${PORT}`));
